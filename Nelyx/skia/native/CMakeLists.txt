cmake_minimum_required(VERSION 3.22)
project(nelyx_skia LANGUAGES CXX)
if(NOT WIN32 AND NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "This standalone artifact supports explicit Windows and Linux SDK profiles")
endif()

set(CMAKE_CXX_STANDARD 20)
set(NELYX_SKIA_SDK "" CACHE FILEPATH "Hash-verified imported SDK targets generated by prepare-artifacts.py")
if(NOT EXISTS "${NELYX_SKIA_SDK}")
  message(FATAL_ERROR "Prepare an explicitly pinned SDK and set NELYX_SKIA_SDK to its sdkCmake result. No automatic vendor build.")
endif()
include("${NELYX_SKIA_SDK}")
if(WIN32 AND NOT NELYX_SKIA_SDK_PROFILE STREQUAL "windows-x64-graphite-text-probe")
  message(FATAL_ERROR "Explicit complete Graphite Vulkan text SDK required.")
endif()
if(CMAKE_SYSTEM_NAME STREQUAL "Linux" AND (NOT NELYX_SKIA_SDK_PROFILE STREQUAL "linux-x64-graphite-text-probe" OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|amd64|AMD64)$"))
  message(FATAL_ERROR "Linux requires the explicit pinned x86_64 complete Graphite/text candidate; other triplets remain unqualified")
endif()
add_subdirectory(core)
add_library(nelyx_skia SHARED adapters/cabi/cabi.cpp rendering/renderer.cpp)
target_include_directories(nelyx_skia PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/rendering")
target_compile_definitions(nelyx_skia PRIVATE NELYX_SKIA_BUILD)
if(WIN32)
  target_sources(nelyx_skia PRIVATE platforms/windows/font_services.cpp)
  target_compile_definitions(nelyx_skia PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
else()
  target_sources(nelyx_skia PRIVATE platforms/linux/font_services.cpp)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(NELYX_FREETYPE REQUIRED IMPORTED_TARGET freetype2)
  target_link_options(nelyx_skia PRIVATE LINKER:--no-undefined)
endif()
if(NELYX_SKIA_SDK_RENDERER STREQUAL "skia-graphite")
  target_sources(nelyx_skia PRIVATE platforms/vulkan/device.cpp platforms/vulkan/gpu_context.cpp)
  target_compile_definitions(nelyx_skia PRIVATE NELYX_GRAPHITE_VULKAN SK_GRAPHITE SK_VULKAN SK_USE_INTERNAL_VULKAN_HEADERS VK_NO_PROTOTYPES)
  if(WIN32)
    target_compile_definitions(nelyx_skia PRIVATE VK_USE_PLATFORM_WIN32_KHR)
  else()
    pkg_check_modules(NELYX_X11 REQUIRED IMPORTED_TARGET x11)
    pkg_check_modules(NELYX_WAYLAND REQUIRED IMPORTED_TARGET wayland-client)
    target_compile_definitions(nelyx_skia PRIVATE VK_USE_PLATFORM_XLIB_KHR VK_USE_PLATFORM_WAYLAND_KHR)
    target_link_libraries(nelyx_skia PRIVATE PkgConfig::NELYX_X11 PkgConfig::NELYX_WAYLAND)
  endif()
endif()
target_link_libraries(nelyx_skia PRIVATE nelyx_skia_core ${NELYX_SKIA_VENDOR_LIBRARIES})
if(WIN32)
  target_link_libraries(nelyx_skia PRIVATE gdi32 dwrite ole32 uuid usp10)
else()
  target_link_libraries(nelyx_skia PRIVATE PkgConfig::NELYX_FREETYPE dl pthread)
endif()
if(MSVC)
  target_compile_options(nelyx_skia PRIVATE /W4 /EHsc /utf-8)
  set_property(TARGET nelyx_skia PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
endif()
