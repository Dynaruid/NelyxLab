cmake_minimum_required(VERSION 3.22)
if(NOT DEFINED NELYX_SKIA_HEADERS)
  message(FATAL_ERROR "NELYX_SKIA_HEADERS must identify the pinned Skia header tree")
endif()
add_library(nelyx_skia_core STATIC core.cpp paragraph_queries.cpp input_geometry.cpp scene_packet.cpp scene_composition.cpp semantics_geometry.cpp)
target_compile_features(nelyx_skia_core PUBLIC cxx_std_20)
target_include_directories(nelyx_skia_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" "${NELYX_SKIA_HEADERS}")
set_property(TARGET nelyx_skia_core PROPERTY POSITION_INDEPENDENT_CODE ON)
if(MSVC)
  set_source_files_properties(input_geometry.cpp semantics_geometry.cpp PROPERTIES COMPILE_OPTIONS "/fp:strict")
else()
  set_source_files_properties(input_geometry.cpp semantics_geometry.cpp PROPERTIES COMPILE_OPTIONS "-fno-fast-math;-ffp-contract=off")
endif()
if(MSVC)
  set_property(TARGET nelyx_skia_core PROPERTY MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
endif()
