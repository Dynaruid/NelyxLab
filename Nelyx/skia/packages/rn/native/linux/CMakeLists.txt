cmake_minimum_required(VERSION 3.25)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT TARGET react_native_linux)
  message(FATAL_ERROR "Linux Skia runtime services require the owning RNL host target")
endif()
if(NOT TARGET nelyx_skia_rn_vulkan)
  add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../vulkan" "${CMAKE_CURRENT_BINARY_DIR}/vulkan")
endif()
find_package(PkgConfig REQUIRED)
pkg_check_modules(NELYX_SKIA_GTK REQUIRED IMPORTED_TARGET gtk4>=4.14)
pkg_check_modules(NELYX_SKIA_DRM REQUIRED IMPORTED_TARGET libdrm)
pkg_check_modules(NELYX_SKIA_SOUP REQUIRED IMPORTED_TARGET libsoup-3.0)
add_library(nelyx_skia_rn_linux INTERFACE)
target_include_directories(nelyx_skia_rn_linux INTERFACE "${CMAKE_CURRENT_LIST_DIR}" "${REACT_NATIVE_LINUX_DIR}/src")
target_link_libraries(nelyx_skia_rn_linux INTERFACE nelyx_skia_rn_vulkan react_native_linux PkgConfig::NELYX_SKIA_GTK PkgConfig::NELYX_SKIA_DRM PkgConfig::NELYX_SKIA_SOUP)
