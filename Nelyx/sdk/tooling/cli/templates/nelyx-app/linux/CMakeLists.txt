cmake_minimum_required(VERSION 3.25)
project(nelyx_linux_app LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|amd64|AMD64)$")
  message(FATAL_ERROR "This native recipe requires Linux x86_64")
endif()
set(NELYX_RUNNER_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/runner" CACHE PATH "Exact independent installed runner or verified build snapshot")
set(REACT_NATIVE_ROOT "${NELYX_RUNNER_ROOT}/node_modules/react-native" CACHE PATH "Independent RN cohort" FORCE)
set(REACT_NATIVE_LINUX_DIR "${NELYX_RUNNER_ROOT}/node_modules/@lucid-softworks/react-native-linux/vnext" CACHE PATH "Owned RNL native source" FORCE)
set(NELYX_REACT_NATIVE_ROOT "${REACT_NATIVE_ROOT}")
set(BUILD_SHARED_LIBS ON CACHE BOOL "Pinned shared runtime libraries" FORCE)
set(FOLLY_TAG "v2024.11.18.00" CACHE STRING "Pinned RNL Folly source" FORCE)
set(FMT_TAG "10.2.1" CACHE STRING "Pinned RNL fmt source" FORCE)
set(CMAKE_DISABLE_FIND_PACKAGE_folly TRUE)
# Provide the pinned fmt target before Folly configures its dependencies. This
# also supports a fresh cache without a system fmt installation.
include(FetchContent)
FetchContent_Declare(fmt GIT_REPOSITORY https://github.com/fmtlib/fmt.git GIT_TAG "${FMT_TAG}" GIT_SHALLOW TRUE)
set(FMT_DOC OFF CACHE BOOL "" FORCE)
set(FMT_TEST OFF CACHE BOOL "" FORCE)
FetchContent_MakeAvailable(fmt)
set(fmt_DIR "${fmt_BINARY_DIR}" CACHE PATH "Pinned source-built fmt config" FORCE)
set(HERMES_COMMIT "e0fc67142ec0763c6b6153ca2bf96df815539782" CACHE STRING "Pinned RNL Hermes source" FORCE)
set(REACT_NATIVE_LINUX_USE_SYSTEM_HERMES OFF CACHE BOOL "Pinned source build required" FORCE)
set(NELYX_SKIA_SDK "" CACHE FILEPATH "Explicit verified native SDK recipe")
set(NELYX_SKIA_RN_MODULES_CMAKE "" CACHE FILEPATH "Explicit verified module source exception")
if(NOT EXISTS "${NELYX_SKIA_SDK}" OR NOT EXISTS "${NELYX_SKIA_RN_MODULES_CMAKE}" OR NOT EXISTS "${NELYX_RUNNER_ROOT}/app-config.json")
  message(FATAL_ERROR "Prepare the app/independent runner and explicit Linux native inputs before configuring")
endif()
file(READ "${NELYX_RUNNER_ROOT}/app-config.json" NELYX_APP_CONFIG)
string(JSON NELYX_APP_NAME GET "${NELYX_APP_CONFIG}" name)
string(JSON NELYX_WINDOW_SYSTEM GET "${NELYX_APP_CONFIG}" windowSystem)
if(NOT NELYX_APP_NAME MATCHES "^[A-Z][A-Za-z0-9]*$" OR NOT NELYX_WINDOW_SYSTEM MATCHES "^(x11|wayland)$")
  message(FATAL_ERROR "Invalid generated Linux app name/window system")
endif()
configure_file(AppConfig.h.in "${CMAKE_CURRENT_BINARY_DIR}/NelyxAppConfig.h" @ONLY)
add_subdirectory("${REACT_NATIVE_LINUX_DIR}" react_native_linux_build)
get_target_property(NELYX_FOLLY_OWNER Folly::folly ALIASED_TARGET)
get_target_property(NELYX_FMT_OWNER fmt::fmt ALIASED_TARGET)
if(NOT NELYX_FOLLY_OWNER STREQUAL "folly" OR NOT NELYX_FMT_OWNER STREQUAL "fmt")
  message(FATAL_ERROR "Folly/fmt targets must belong to the pinned source builds")
endif()
FetchContent_GetProperties(hermes SOURCE_DIR NELYX_HERMES_SOURCE)
execute_process(COMMAND git -C "${NELYX_HERMES_SOURCE}" rev-parse HEAD RESULT_VARIABLE NELYX_HERMES_HEAD_RESULT OUTPUT_VARIABLE NELYX_HERMES_HEAD OUTPUT_STRIP_TRAILING_WHITESPACE)
execute_process(COMMAND git -C "${NELYX_HERMES_SOURCE}" status --porcelain --untracked-files=normal RESULT_VARIABLE NELYX_HERMES_STATE_RESULT OUTPUT_VARIABLE NELYX_HERMES_STATE OUTPUT_STRIP_TRAILING_WHITESPACE)
if(NOT NELYX_HERMES_HEAD_RESULT EQUAL 0 OR NOT NELYX_HERMES_STATE_RESULT EQUAL 0 OR NOT NELYX_HERMES_HEAD STREQUAL HERMES_COMMIT OR NOT NELYX_HERMES_STATE STREQUAL "")
  message(FATAL_ERROR "The actual Hermes source must be the clean pinned checkout, including cached FetchContent overrides")
endif()
configure_file("${NELYX_HERMES_SOURCE}/LICENSE" "${CMAKE_CURRENT_BINARY_DIR}/nelyx-hermes-LICENSE" COPYONLY)
foreach(NELYX_DEPENDENCY folly fmt)
  FetchContent_GetProperties(${NELYX_DEPENDENCY} SOURCE_DIR NELYX_DEPENDENCY_SOURCE)
  if(NELYX_DEPENDENCY STREQUAL "folly")
    set(NELYX_DEPENDENCY_PIN "0660ead5b610047a279cc291129e98e3f2a1fc37")
  else()
    set(NELYX_DEPENDENCY_PIN "e69e5f977d458f2650bb346dadf2ad30c5320281")
  endif()
  execute_process(COMMAND git -C "${NELYX_DEPENDENCY_SOURCE}" rev-parse HEAD RESULT_VARIABLE NELYX_DEPENDENCY_HEAD_RESULT OUTPUT_VARIABLE NELYX_DEPENDENCY_HEAD OUTPUT_STRIP_TRAILING_WHITESPACE)
  execute_process(COMMAND git -C "${NELYX_DEPENDENCY_SOURCE}" status --porcelain --untracked-files=normal RESULT_VARIABLE NELYX_DEPENDENCY_STATE_RESULT OUTPUT_VARIABLE NELYX_DEPENDENCY_STATE OUTPUT_STRIP_TRAILING_WHITESPACE)
  if(NOT NELYX_DEPENDENCY_HEAD_RESULT EQUAL 0 OR NOT NELYX_DEPENDENCY_STATE_RESULT EQUAL 0 OR NOT NELYX_DEPENDENCY_HEAD STREQUAL NELYX_DEPENDENCY_PIN OR NOT NELYX_DEPENDENCY_STATE STREQUAL "")
    message(FATAL_ERROR "${NELYX_DEPENDENCY} requires its clean pinned source exception; unverified system libraries are not admitted")
  endif()
  configure_file("${NELYX_DEPENDENCY_SOURCE}/LICENSE" "${CMAKE_CURRENT_BINARY_DIR}/nelyx-${NELYX_DEPENDENCY}-LICENSE" COPYONLY)
endforeach()
configure_file(NativeSources.json.in "${CMAKE_CURRENT_BINARY_DIR}/nelyx-native-source.json" @ONLY)
add_subdirectory("${NELYX_RUNNER_ROOT}/node_modules/@nelyx/skia-rn/native/linux" skia-linux)
foreach(NELYX_RUNTIME_TARGET react_native_linux hermesvm libhermes hermes jsi folly fmt)
  if(TARGET ${NELYX_RUNTIME_TARGET})
    get_target_property(NELYX_RUNTIME_TYPE ${NELYX_RUNTIME_TARGET} TYPE)
    if(NELYX_RUNTIME_TYPE STREQUAL "SHARED_LIBRARY")
      set_target_properties(${NELYX_RUNTIME_TARGET} PROPERTIES BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "$ORIGIN")
    endif()
  endif()
endforeach()
add_executable(rn-linux-app main.cpp)
target_include_directories(rn-linux-app PRIVATE "${NELYX_RUNNER_ROOT}/node_modules/@nelyx/sdk/native/linux" "${CMAKE_CURRENT_BINARY_DIR}" "$<TARGET_PROPERTY:react_native_rn_renderer,INTERFACE_INCLUDE_DIRECTORIES>")
target_compile_options(rn-linux-app PRIVATE -Wno-changes-meaning)
target_link_libraries(rn-linux-app PRIVATE nelyx_skia_rn_linux Folly::folly Hermes::Hermes)
set_target_properties(rn-linux-app PROPERTIES BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "$ORIGIN")
install(TARGETS rn-linux-app RUNTIME DESTINATION bin)
